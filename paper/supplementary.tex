    \label{tab:forbidden}
    \begin{tabular}{@{}l p{11.5cm}@{}}
        \toprule
        Column & Reason \\
        \midrule
        \texttt{DOI}, \texttt{doi\_group} & publication identity; used only to define split groups \\
        \texttt{Value}, \texttt{Raw\_value} & the target and its text \\
        \texttt{Raw\_unit}, \texttt{Unit} & derived from the target (unit as written next to the value) \\
        \texttt{Property} & selects the target (one model per property) \\
        \texttt{Title} & publication identity; admitted only as keyword flags in $t_3$ \\
        \texttt{Journal} & publication identity and reporting venue \\
        \texttt{Date} & publication identity and time; used only to order publications in the temporal split \\
        \texttt{Correctness} & post-hoc manual evaluation label \\
        \texttt{Num\_records} & merged-file count (not in \texttt{battery.csv}; listed for completeness) \\
        \texttt{Warning} = S & series membership; a function of within-paper multiplicity (identity proxy) \\
        \texttt{Warning} = L & deterministic function of the target value (near-limit ranges) \\
        \texttt{Tag} & Calculated records are derived and filtered out; constant otherwise \\
        \texttt{Extracted\_name} (raw) & admitted only through parsed composition and elemental-property features \\
        \texttt{Name} (raw) & admitted only as name flags (\texttt{txt\_}) and, in $t_{2,\mathrm{id}}$, as \texttt{id\_name} \\
        \texttt{Specifier} (raw) & admitted only as keyword flags \\
        \texttt{Info} (raw) & admitted only as parsed current density, C-rate and cycle number \\
        \bottomrule
    \end{tabular}
